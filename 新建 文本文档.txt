第一步：连接服务器

  ssh root@你的服务器IP

  ---
  第二步：安装环境依赖

  # 更新系统
  apt update && apt upgrade -y

  # 安装工具
  apt install -y curl wget git unzip vim net-tools

  # 安装 Python 3.10
  add-apt-repository ppa:deadsnakes/ppa -y
  apt update
  apt install -y python3.10 python3.10-venv python3.10-dev python3-pip

  # 安装 Node.js 18
  curl -fsSL https://deb.nodesource.com/setup_18.x | bash -
  apt install -y nodejs

  # 安装 MySQL
  apt install -y mysql-server
  systemctl start mysql && systemctl enable mysql
  mysql_secure_installation   # 按提示设置 root 密码

  # 安装 Nginx
  apt install -y nginx
  systemctl start nginx && systemctl enable nginx

  # 安装 PM2（管理后端进程）
  npm install -g pm2

  # 安装中文字体（水印功能需要）
  apt install -y fonts-wqy-zenhei fonts-wqy-microhei

  ---
  第三步：上传项目文件

  用 FileZilla（SFTP，端口22）将本机D:\维修记录web项目 整个目录上传到服务器 /opt/repair-system/。

  上传完后验证：
  ls /opt/repair-system/
  # 应看到 backend、frontend、database 等目录

  ---
  第四步：导入数据库

  mysql -u root -p
  # 输入密码后执行：
  source /opt/repair-system/database/repair_db.sql
  USE repair_db;
  SHOW TABLES;   # 验证表是否存在
  EXIT;

  ---
  第五步：部署后端

  cd /opt/repair-system/backend

  # 创建虚拟环境并安装依赖
  python3.10 -m venv venv
  source venv/bin/activate
  pip install --upgrade pip
  pip install -r requirements.txt
  deactivate

  # 创建配置文件
  cp .env.example .env
  nano .env    # 填入数据库密码等配置

  .env 内容：
  DB_HOST=localhost
  DB_PORT=3306
  DB_USER=root
  DB_PASSWORD=你设置的MySQL密码
  DB_NAME=repair_db
  UPLOAD_DIR=uploads
  MAX_FILE_SIZE=10485760

  # 创建上传目录
  mkdir -p uploads/exports
  chmod 755 uploads

  # 用 PM2 启动后端
  pm2 start "/opt/repair-system/backend/venv/bin/python -m uvicorn main:app --host 127.0.0.1 --port 8000" \--name
  repair-api \
    --cwd /opt/repair-system/backend

  # 设置开机自启
  pm2 save && pm2 startup
  # 复制 pm2 startup 输出的那条命令执行

  # 验证
  curl http://127.0.0.1:8000/health
  # 应返回 {"status":"ok","version":"2.0.0"}

  ---
  第六步：构建并部署前端

  cd /opt/repair-system/frontend
  npm install
  npm run build

  # 复制到Nginx 目录
  mkdir -p /var/www/repair-system
  cp -r dist/* /var/www/repair-system/
  chown -R www-data:www-data /var/www/repair-system

  ---
  第七步：配置 Nginx

  nano /etc/nginx/sites-available/repair-system

  粘贴以下内容（替换 你的服务器IP）：

  server {
      listen 80;
      server_name 你的服务器IP;
      charset utf-8;

      root /var/www/repair-system;
      index index.html;

      location / {
          try_files $uri $uri/ /index.html;
          expires 1d;
      }

      location /api/ {
          proxy_pass http://127.0.0.1:8000;
          proxy_set_header Host $host;
          proxy_set_header X-Real-IP $remote_addr;
          proxy_read_timeout 120s;
          client_max_body_size 20M;
      }

      location /uploads/ {
          proxy_pass http://127.0.0.1:8000;
          expires 7d;
      }
  }

  # 启用站点并重载
  ln -s /etc/nginx/sites-available/repair-system /etc/nginx/sites-enabled/
  nginx -t          # 检查语法
  systemctl reload nginx

  ---
  第八步：开放防火墙

  ufw allow 22/tcp
  ufw allow 80/tcp
  ufw allow 443/tcp
  ufw enable

  ▎ ⚠️ 如果是阿里云/腾讯云，还需要在云控制台的「安全组」里手动开放 80 端口。

  ---
  第九步：验证

  curl http://127.0.0.1:8000/health   # 后端
  curl http://localhost/               # 前端
  pm2 list                # 进程状态

  浏览器访问 http://你的服务器IP，看到登录页即成功。

  默认账号： admin / admin123（登录后请立即改密码）

  ---
  常用维护命令

  pm2 restart repair-api    # 重启后端
  pm2 logs repair-api       # 查看后端日志
  systemctl reload nginx    # 重载 Nginx
  systemctl restart mysql# 重启 MySQL
